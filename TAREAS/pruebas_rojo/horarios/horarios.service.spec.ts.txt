import { BadRequestException, ConflictException } from "@nestjs/common";
import { AnioLectivoCls } from "@/common/anio-lectivo";
import { AuditService } from "@/common/services/audit.service";
import { AnioAcademicoResponseDto } from "@/features/anio-escolar/dto";
import { AniosAcademicosService } from "@/features/anio-escolar/anios.service";
import { DocentesRepository } from "@/features/docentes";
import { SeccionResponseDto, SeccionesRepository } from "@/features/secciones";
import {
  BloqueHorarioResponseDto,
  CreateBloqueHorarioDto,
  HorariosRepository,
  HorariosService,
  UpdateBloqueHorarioDto,
} from "@/features/horarios";

describe("RF-008: Horario Semanal — Núcleo (HU3: Bloques)", () => {
  let service: HorariosService;
  let mockHorariosRepo: jest.Mocked<Partial<HorariosRepository>>;
  let mockSeccionesRepo: jest.Mocked<Partial<SeccionesRepository>>;
  let mockDocentesRepo: jest.Mocked<Partial<DocentesRepository>>;
  let mockAniosService: jest.Mocked<Partial<AniosAcademicosService>>;
  let mockAnioLectivoCls: jest.Mocked<Partial<AnioLectivoCls>>;
  let mockAuditService: jest.Mocked<Partial<AuditService>>;

  const mockAnio2027: AnioAcademicoResponseDto = {
    id: "anio-2027",
    anio: 2027,
    nombre: "Año Escolar 2027",
    fechaInicio: "2027-03-01",
    fechaFin: "2027-12-20",
    estado: "en_curso",
    reabiertoHasta: null,
    motivoReapertura: null,
    bimestres: [],
  };

  const mockSeccion5A: SeccionResponseDto = {
    id: "sec-5a",
    anioAcademicoId: "anio-2027",
    gradoId: "grado-5",
    gradoNumero: 5,
    gradoNombre: "5.° de secundaria",
    nombre: "A",
    aula: "Pab. B – 201",
    vacantes: 35,
    estado: "activa",
    tutor: null,
    matriculados: 0,
    totalCursos: 12,
    cursosSinDocente: 0,
  };

  const mockBloqueMatematica: BloqueHorarioResponseDto = {
    id: "bloque-1",
    seccionId: "sec-5a",
    cursoSeccionId: "cs-mat-5a",
    cursoNombre: "Matemática",
    colorHex: "#2563EB",
    docenteNombre: "Luis Quispe Mamani",
    diaSemana: 1, // Lunes
    horaInicio: "07:45",
    horaFin: "08:30",
    tipoBloque: "clase",
    aula: "Pab. B – 201",
  };

  beforeEach(() => {
    mockAuditService = {
      log: jest.fn().mockResolvedValue(undefined),
    };

    mockAnioLectivoCls = {
      id: jest.fn().mockReturnValue("anio-2027"),
    };

    mockAniosService = {
      findById: jest.fn().mockResolvedValue(mockAnio2027),
      findEnCurso: jest.fn().mockResolvedValue(mockAnio2027),
    };

    mockDocentesRepo = {
      findById: jest.fn().mockResolvedValue(null),
    };

    mockSeccionesRepo = {
      findById: jest.fn().mockImplementation((id: string) => {
        if (id === "sec-5a") return Promise.resolve(mockSeccion5A);
        return Promise.resolve(null);
      }),
    };

    mockHorariosRepo = {
      findBySeccion: jest.fn().mockResolvedValue([mockBloqueMatematica]),
      findById: jest.fn().mockImplementation((id: string) => {
        if (id === "bloque-1") return Promise.resolve(mockBloqueMatematica);
        return Promise.resolve(null);
      }),
      findSolapados: jest.fn().mockResolvedValue([]),
      findCruceDocente: jest.fn().mockResolvedValue(null),
      horasPorCurso: jest.fn().mockResolvedValue({ "cs-mat-5a": 6 }),
      horasPlanPorCurso: jest.fn().mockResolvedValue({
        "cs-mat-5a": { cursoNombre: "Matemática", horasPlan: 6 },
      }),
      create: jest.fn().mockResolvedValue(mockBloqueMatematica),
      update: jest.fn().mockResolvedValue(mockBloqueMatematica),
      delete: jest.fn().mockResolvedValue(true),
      tieneSesiones: jest.fn().mockResolvedValue(false),
      findByDocente: jest.fn().mockResolvedValue([]),
    };

    service = new HorariosService(
      mockHorariosRepo as unknown as HorariosRepository,
      mockSeccionesRepo as unknown as SeccionesRepository,
      mockDocentesRepo as unknown as DocentesRepository,
      mockAniosService as unknown as AniosAcademicosService,
      mockAnioLectivoCls as unknown as AnioLectivoCls,
      mockAuditService as unknown as AuditService,
    );
  });

  describe("Creación y edición de bloques", () => {
    it("CP-RF008-12: debe registrar un bloque de clase con curso y docente y auditar la mutación", async () => {
      // Dado que el lunes 07:45–08:30 de 5.° A está libre
      const dto: CreateBloqueHorarioDto = {
        cursoSeccionId: "cs-mat-5a",
        diaSemana: 1,
        horaInicio: "07:45",
        horaFin: "08:30",
        tipoBloque: "clase",
        aula: "Pab. B – 201",
      };

      // Cuando le asigno Matemática
      const resultado = await service.crearBloque("sec-5a", dto, "user-admin", "192.168.1.30");

      // Entonces el bloque queda como clase de Matemática con su docente
      expect(resultado.bloque.cursoNombre).toBe("Matemática");
      expect(resultado.bloque.docenteNombre).toBe("Luis Quispe Mamani");
      expect(resultado.bloque.horaInicio).toBe("07:45");
      expect(resultado.bloque.horaFin).toBe("08:30");

      expect(mockHorariosRepo.create).toHaveBeenCalledWith(
        expect.objectContaining({
          seccionId: "sec-5a",
          cursoSeccionId: "cs-mat-5a",
          diaSemana: 1,
          horaInicio: "07:45",
          horaFin: "08:30",
        }),
      );

      // Y audita la creación
      expect(mockAuditService.log).toHaveBeenCalledWith(
        expect.objectContaining({
          usuarioId: "user-admin",
          accion: "crear",
          tabla: "horarios",
          registroId: "bloque-1",
          direccionIp: "192.168.1.30",
        }),
      );
    });

    it("CP-RF008-13: debe guardar un bloque doble de 90 minutos como un solo bloque de 2 horas pedagógicas", async () => {
      // Dado que el lunes 08:30–10:00 está libre
      const dtoDoble: CreateBloqueHorarioDto = {
        cursoSeccionId: "cs-com-5a",
        diaSemana: 1,
        horaInicio: "08:30",
        horaFin: "10:00", // 90 min = 2 horas pedagógicas de 45 min
        tipoBloque: "clase",
      };

      const bloqueDoble: BloqueHorarioResponseDto = {
        id: "bloque-doble",
        seccionId: "sec-5a",
        cursoSeccionId: "cs-com-5a",
        cursoNombre: "Comunicación",
        diaSemana: 1,
        horaInicio: "08:30",
        horaFin: "10:00",
        tipoBloque: "clase",
        aula: null,
      };
      mockHorariosRepo.create = jest.fn().mockResolvedValue(bloqueDoble);

      // Cuando registro un bloque de Comunicación de 08:30 a 10:00
      const resultado = await service.crearBloque("sec-5a", dtoDoble, "user-admin");

      // Entonces el sistema lo guarda como un solo bloque
      expect(resultado.bloque.horaInicio).toBe("08:30");
      expect(resultado.bloque.horaFin).toBe("10:00");
      expect(mockHorariosRepo.create).toHaveBeenCalled();
    });

    it("CP-RF008-14: debe impedir guardar si el bloque se superpone con otro bloque de la sección", async () => {
      // Dado que 5.° A tiene clase el lunes de 08:30 a 09:15
      mockHorariosRepo.findSolapados = jest.fn().mockResolvedValue([mockBloqueMatematica]);

      const dtoSolapado: CreateBloqueHorarioDto = {
        cursoSeccionId: "cs-com-5a",
        diaSemana: 1,
        horaInicio: "09:00",
        horaFin: "09:45",
        tipoBloque: "clase",
      };

      // Cuando registro otro bloque el lunes de 09:00 a 09:45
      // Entonces el sistema impide guardar y muestra "El bloque se cruza con otro bloque de la sección."
      await expect(service.crearBloque("sec-5a", dtoSolapado, "user-admin")).rejects.toThrow(
        new ConflictException("El bloque se cruza con otro bloque de la sección."),
      );
      expect(mockHorariosRepo.create).not.toHaveBeenCalled();
    });

    it("CP-RF008-15: debe bloquear si el docente tiene cruce con otra sección del año a la misma hora", async () => {
      // Dado que el docente Quispe dicta en 4.° B el lunes de 08:30 a 09:15
      mockHorariosRepo.findCruceDocente = jest.fn().mockResolvedValue({
        docenteNombre: "Luis Quispe Mamani",
        gradoNumero: 4,
        seccionNombre: "B",
        diaSemana: 1,
        horaInicio: "08:30",
        horaFin: "09:15",
      });

      const dtoCruce: CreateBloqueHorarioDto = {
        cursoSeccionId: "cs-mat-5a",
        diaSemana: 1,
        horaInicio: "08:30",
        horaFin: "09:15",
        tipoBloque: "clase",
      };

      // Cuando asigno su curso en 5.° A el lunes de 08:30 a 09:15
      // Entonces el sistema impide guardar y muestra "El docente ya dicta en 4.° B el lunes de 08:30 a 09:15."
      await expect(service.crearBloque("sec-5a", dtoCruce, "user-admin")).rejects.toThrow(
        new ConflictException("El docente ya dicta en 4.° B el lunes de 08:30 a 09:15."),
      );
      expect(mockHorariosRepo.create).not.toHaveBeenCalled();
    });

    it("CP-RF008-16: debe guardar y advertir si las horas del horario difieren de las horas del plan de estudios obtenidas del repositorio", async () => {
      // Dado que Matemática tiene 6 horas en el plan de estudios (según horasPlanPorCurso del repositorio) y 5 en el horario
      mockHorariosRepo.horasPlanPorCurso = jest.fn().mockResolvedValue({
        "cs-mat-5a": { cursoNombre: "Matemática", horasPlan: 6 },
      });
      mockHorariosRepo.horasPorCurso = jest.fn().mockResolvedValue({
        "cs-mat-5a": 5, // 5 horas frente a 6 del plan
      });

      const dto: CreateBloqueHorarioDto = {
        cursoSeccionId: "cs-mat-5a",
        diaSemana: 1,
        horaInicio: "07:45",
        horaFin: "08:30",
        tipoBloque: "clase",
      };

      // Cuando guardo un horario con 5 horas de Matemática
      const resultado = await service.crearBloque("sec-5a", dto, "user-admin");

      // Entonces el sistema consulta las horas del plan al repositorio
      expect(mockHorariosRepo.horasPlanPorCurso).toHaveBeenCalledWith("sec-5a");

      // Y guarda mostrando la advertencia oficial sin bloquear
      expect(resultado.bloque).toBeDefined();
      expect(resultado.advertencias).toContain(
        "Matemática tiene 5 horas en el horario y 6 en el plan de estudios.",
      );
      expect(mockHorariosRepo.create).toHaveBeenCalled();
    });

    it("CP-RF008-16: no debe generar advertencia si las horas del horario coinciden con las horas del plan de estudios", async () => {
      // Dado que Matemática tiene 6 horas en el plan y 6 horas programadas en el horario
      mockHorariosRepo.horasPlanPorCurso = jest.fn().mockResolvedValue({
        "cs-mat-5a": { cursoNombre: "Matemática", horasPlan: 6 },
      });
      mockHorariosRepo.horasPorCurso = jest.fn().mockResolvedValue({
        "cs-mat-5a": 6,
      });

      const dto: CreateBloqueHorarioDto = {
        cursoSeccionId: "cs-mat-5a",
        diaSemana: 1,
        horaInicio: "07:45",
        horaFin: "08:30",
        tipoBloque: "clase",
      };

      // Cuando guardo el bloque
      const resultado = await service.crearBloque("sec-5a", dto, "user-admin");

      // Entonces consulta las horas del plan y no genera advertencia para Matemática
      expect(mockHorariosRepo.horasPlanPorCurso).toHaveBeenCalledWith("sec-5a");
      expect(resultado.bloque).toBeDefined();
      expect(resultado.advertencias).not.toContain(
        expect.stringContaining("Matemática tiene"),
      );
    });

    it("CP-RF008-16: debe obtener las horas y nombre del plan desde el repositorio y no deducirlos del nombre del curso (ej. Taller de Robótica)", async () => {
      // Dado un curso no convencional "Taller de Robótica" con 2 horas en el plan de estudios
      const mockBloqueRobotica: BloqueHorarioResponseDto = {
        ...mockBloqueMatematica,
        id: "bloque-rob-1",
        cursoSeccionId: "cs-rob-5a",
        cursoNombre: "Taller de Robótica",
      };
      mockHorariosRepo.create = jest.fn().mockResolvedValue(mockBloqueRobotica);
      mockHorariosRepo.horasPlanPorCurso = jest.fn().mockResolvedValue({
        "cs-rob-5a": { cursoNombre: "Taller de Robótica", horasPlan: 2 },
      });
      mockHorariosRepo.horasPorCurso = jest.fn().mockResolvedValue({
        "cs-rob-5a": 3, // 3 horas en el horario frente a 2 del plan
      });

      const dtoRobotica: CreateBloqueHorarioDto = {
        cursoSeccionId: "cs-rob-5a",
        diaSemana: 2,
        horaInicio: "08:30",
        horaFin: "09:15",
        tipoBloque: "clase",
      };

      // Cuando registro el bloque de Taller de Robótica
      const resultado = await service.crearBloque("sec-5a", dtoRobotica, "user-admin");

      // Entonces el sistema debe haber consultado el repositorio y mostrar el nombre y horas exactos del plan
      expect(mockHorariosRepo.horasPlanPorCurso).toHaveBeenCalledWith("sec-5a");
      expect(resultado.bloque).toBeDefined();
      expect(resultado.advertencias).toContain(
        "Taller de Robótica tiene 3 horas en el horario y 2 en el plan de estudios.",
      );
      expect(resultado.advertencias).not.toContain(
        expect.stringContaining("Matemática"),
      );
    });

    it("CP-RF008-18: con el año en curso, debe retornar mensaje de vigencia desde mañana", async () => {
      // Dado que el año está en curso
      const dtoUpdate: UpdateBloqueHorarioDto = {
        horaInicio: "10:15",
        horaFin: "11:00",
      };

      // Cuando cambio un bloque y guardo
      const resultado = await service.actualizarBloque("bloque-1", dtoUpdate, "user-admin");

      // Entonces el sistema muestra "El horario se guardó. Los cambios rigen desde mañana."
      expect(resultado.mensaje).toBe("El horario se guardó. Los cambios rigen desde mañana.");
      expect(mockHorariosRepo.update).toHaveBeenCalledWith("bloque-1", dtoUpdate);
      expect(mockAuditService.log).toHaveBeenCalledWith(
        expect.objectContaining({
          usuarioId: "user-admin",
          accion: "actualizar",
          tabla: "horarios",
          registroId: "bloque-1",
        }),
      );
    });

    it("debe rechazar si la hora de fin es anterior o igual a la de inicio", async () => {
      const dtoInvertido: CreateBloqueHorarioDto = {
        diaSemana: 1,
        horaInicio: "10:00",
        horaFin: "09:15",
        tipoBloque: "clase",
        cursoSeccionId: "cs-mat-5a",
      };

      await expect(service.crearBloque("sec-5a", dtoInvertido, "user-admin")).rejects.toThrow(
        new BadRequestException("La hora de fin debe ser posterior a la hora de inicio."),
      );

      const dtoMismaHora: CreateBloqueHorarioDto = {
        diaSemana: 1,
        horaInicio: "10:00",
        horaFin: "10:00",
        tipoBloque: "clase",
        cursoSeccionId: "cs-mat-5a",
      };

      await expect(service.crearBloque("sec-5a", dtoMismaHora, "user-admin")).rejects.toThrow(
        new BadRequestException("La hora de fin debe ser posterior a la hora de inicio."),
      );
    });

    it("debe listar los bloques de la sección con resumen de horas", async () => {
      const resultado = await service.listarPorSeccion("sec-5a");

      expect(resultado.bloques).toHaveLength(1);
      expect(mockHorariosRepo.findBySeccion).toHaveBeenCalledWith("sec-5a");
      expect(mockHorariosRepo.horasPlanPorCurso).toHaveBeenCalledWith("sec-5a");
    });
  });

  describe("Eliminación de bloques", () => {
    it("debe bloquear la eliminación si el bloque ya cuenta con sesiones de clase registradas", async () => {
      mockHorariosRepo.tieneSesiones = jest.fn().mockResolvedValue(true);

      await expect(service.eliminarBloque("bloque-1", "user-admin")).rejects.toThrow(
        new ConflictException("No se puede eliminar un bloque que ya tiene sesiones de clase."),
      );
      expect(mockHorariosRepo.delete).not.toHaveBeenCalled();
    });

    it("debe eliminar el bloque si no tiene sesiones de clase y registrar auditoría", async () => {
      mockHorariosRepo.tieneSesiones = jest.fn().mockResolvedValue(false);

      await service.eliminarBloque("bloque-1", "user-admin", "192.168.1.40");

      expect(mockHorariosRepo.delete).toHaveBeenCalledWith("bloque-1");
      expect(mockAuditService.log).toHaveBeenCalledWith(
        expect.objectContaining({
          usuarioId: "user-admin",
          accion: "eliminar",
          tabla: "horarios",
          registroId: "bloque-1",
          direccionIp: "192.168.1.40",
        }),
      );
    });
  });
});
