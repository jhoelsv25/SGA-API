import { Reflector } from "@nestjs/core";
import { MENU_HORARIOS } from "@/common/constants";
import { MENU_ACCESS_KEY } from "@/common/decorators";
import { AuthenticatedUser } from "@/features/auth/interfaces/jwt-payload.interface";
import { ClientInfoData } from "@/common/decorators";
import {
  HorariosController,
  HorariosService,
} from "@/features/horarios";

describe("HorariosController — Grilla y Copia (RF-008: Seguridad y Endpoints)", () => {
  let controller: HorariosController;
  let service: jest.Mocked<Partial<HorariosService>>;
  let reflector: Reflector;

  const mockUsuarioAdmin: AuthenticatedUser = {
    id: "user-admin",
    username: "admin",
    role: "Administrador",
    roleId: "rol-admin-uuid",
    personaId: "per-admin-uuid",
  };

  const mockClientInfo: ClientInfoData = {
    ip: "192.168.1.10",
    userAgent: "Mozilla/5.0",
  };

  beforeEach(() => {
    reflector = new Reflector();

    service = {
      proponerGrilla: jest.fn().mockResolvedValue({ horaEntrada: "07:45", bloques: [] }),
      copiar: jest.fn().mockResolvedValue({ bloquesCreados: 5, conflictos: [] }),
    };

    controller = new HorariosController(service as unknown as HorariosService);
  });

  describe("CP-RF008-20: Control de acceso a copia y grilla (@MenuAccess)", () => {
    it("debe exigir el permiso de organización escolar (@MenuAccess('/admin/horarios')) al copiar un horario", () => {
      // Dado el endpoint copiar del controlador
      const rutaPermitida = reflector.get<string>(MENU_ACCESS_KEY, controller.copiar);

      // Entonces debe requerir estrictamente la ruta de MENU_HORARIOS
      expect(rutaPermitida).toBe(MENU_HORARIOS);
      expect(rutaPermitida).toBe("/admin/horarios");
    });

    it("no debe exigir restricción de @MenuAccess en propuesta de grilla (proponerGrilla)", () => {
      // Dado el endpoint de proponerGrilla
      const rutaProponer = reflector.get<string>(MENU_ACCESS_KEY, controller.proponerGrilla);

      // Entonces no debe estar bloqueado por @MenuAccess
      expect(rutaProponer).toBeUndefined();
    });
  });

  describe("Delegación de endpoints de copia y grilla al servicio", () => {
    it("copiar() debe delegar la copia al servicio con los ids de origen y destino", async () => {
      const resultado = await controller.copiar("sec-5a", "sec-5b", mockUsuarioAdmin, mockClientInfo);

      expect(resultado.bloquesCreados).toBe(5);
      expect(service.copiar).toHaveBeenCalledWith(
        "sec-5a",
        "sec-5b",
        mockUsuarioAdmin.id,
        mockClientInfo.ip,
      );
    });

    it("proponerGrilla() debe delegar la grilla al servicio", async () => {
      const resultado = await controller.proponerGrilla("sec-5a", "07:45");

      expect(resultado.horaEntrada).toBe("07:45");
      expect(service.proponerGrilla).toHaveBeenCalledWith("sec-5a", "07:45");
    });
  });
});
