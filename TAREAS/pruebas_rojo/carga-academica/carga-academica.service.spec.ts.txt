import { BadRequestException, ConflictException, NotFoundException } from "@nestjs/common";
import { AnioLectivoCls } from "@/common/anio-lectivo";
import { AuditService } from "@/common/services/audit.service";
import { AnioAcademicoResponseDto } from "@/features/anio-escolar/dto";
import { AniosAcademicosService } from "@/features/anio-escolar/anios.service";
import { DocenteResponseDto, DocentesRepository } from "@/features/docentes";
import {
  AsignarDocenteDto,
  CargaAcademicaItemDto,
  CargaAcademicaRepository,
  CargaAcademicaService,
  CursoDocenteItemDto,
  CursoSeccionDetalle,
} from "@/features/carga-academica";

describe("RF-008: Carga Académica (HU2)", () => {
  let service: CargaAcademicaService;
  let mockCargaRepo: jest.Mocked<Partial<CargaAcademicaRepository>>;
  let mockDocentesRepo: jest.Mocked<Partial<DocentesRepository>>;
  let mockAniosService: jest.Mocked<Partial<AniosAcademicosService>>;
  let mockAnioLectivoCls: jest.Mocked<Partial<AnioLectivoCls>>;
  let mockAuditService: jest.Mocked<Partial<AuditService>>;

  const mockAnio2027: AnioAcademicoResponseDto = {
    id: "anio-2027",
    anio: 2027,
    nombre: "Año Escolar 2027",
    fechaInicio: "2027-03-01",
    fechaFin: "2027-12-20",
    estado: "en_curso",
    reabiertoHasta: null,
    motivoReapertura: null,
    bimestres: [],
  };

  const mockDocenteQuispe: DocenteResponseDto = {
    id: "doc-quispe",
    personaId: "per-quispe",
    codigo: "DOC010",
    persona: {
      id: "per-quispe",
      tipoDocumento: "dni",
      numeroDocumento: "20000010",
      nombres: "Luis",
      apellidoPaterno: "Quispe",
      apellidoMaterno: "Mamani",
      nombreCompleto: "Luis Quispe Mamani",
      sexo: "M",
      correo: "lquispe@colegio.edu.pe",
      esProvisional: false,
      fichas: { esDocente: true, esEstudiante: false, esApoderado: false, tieneUsuario: true },
      creadoEn: "2020-03-01",
      actualizadoEn: "2020-03-01",
    },
    tipoContrato: "nombrado",
    jornadaHoras: 30,
    fechaIngreso: "2020-03-01",
    estado: "activo",
    creadoEn: "2020-03-01",
    actualizadoEn: "2020-03-01",
  };

  const mockDocenteFlores: DocenteResponseDto = {
    id: "doc-flores",
    personaId: "per-flores",
    codigo: "DOC020",
    persona: {
      id: "per-flores",
      tipoDocumento: "dni",
      numeroDocumento: "20000020",
      nombres: "Rosa",
      apellidoPaterno: "Flores",
      apellidoMaterno: "Paredes",
      nombreCompleto: "Rosa Flores Paredes",
      sexo: "F",
      correo: "rflores@colegio.edu.pe",
      esProvisional: false,
      fichas: { esDocente: true, esEstudiante: false, esApoderado: false, tieneUsuario: true },
      creadoEn: "2022-03-01",
      actualizadoEn: "2022-03-01",
    },
    tipoContrato: "contratado",
    jornadaHoras: 30,
    fechaIngreso: "2022-03-01",
    estado: "activo",
    creadoEn: "2022-03-01",
    actualizadoEn: "2022-03-01",
  };

  const mockDocenteCesado: DocenteResponseDto = {
    id: "doc-ramos",
    personaId: "per-ramos",
    codigo: "DOC030",
    persona: {
      id: "per-ramos",
      tipoDocumento: "dni",
      numeroDocumento: "20000030",
      nombres: "Alberto",
      apellidoPaterno: "Ramos",
      apellidoMaterno: "Castillo",
      nombreCompleto: "Alberto Ramos Castillo",
      sexo: "M",
      correo: "aramos@colegio.edu.pe",
      esProvisional: false,
      fichas: { esDocente: true, esEstudiante: false, esApoderado: false, tieneUsuario: false },
      creadoEn: "2019-03-01",
      actualizadoEn: "2025-12-31",
    },
    tipoContrato: "nombrado",
    jornadaHoras: 30,
    fechaIngreso: "2019-03-01",
    fechaCese: "2025-12-31",
    estado: "cesado",
    creadoEn: "2019-03-01",
    actualizadoEn: "2025-12-31",
  };

  const mockCursoSeccionMatematica: CursoSeccionDetalle = {
    id: "cs-mat-5a",
    seccionId: "sec-5a",
    cursoId: "curso-mat",
    cursoNombre: "Matemática",
    docenteId: null,
    horasPlan: 6,
    anioAcademicoId: "anio-2027",
  };

  const mockItemMatematicaAsignado: CargaAcademicaItemDto = {
    id: "cs-mat-5a",
    cursoId: "curso-mat",
    cursoNombre: "Matemática",
    colorHex: "#2563EB",
    horasPlan: 6,
    horasHorario: 6,
    docente: {
      id: "doc-quispe",
      codigo: "DOC010",
      nombreCompleto: "Luis Quispe Mamani",
    },
  };

  beforeEach(() => {
    mockAuditService = {
      log: jest.fn().mockResolvedValue(undefined),
    };

    mockAnioLectivoCls = {
      id: jest.fn().mockReturnValue("anio-2027"),
    };

    mockAniosService = {
      findById: jest.fn().mockResolvedValue(mockAnio2027),
      findEnCurso: jest.fn().mockResolvedValue(mockAnio2027),
    };

    mockDocentesRepo = {
      findById: jest.fn().mockImplementation((id: string) => {
        if (id === "doc-quispe") return Promise.resolve(mockDocenteQuispe);
        if (id === "doc-flores") return Promise.resolve(mockDocenteFlores);
        if (id === "doc-ramos") return Promise.resolve(mockDocenteCesado);
        return Promise.resolve(null);
      }),
    };

    mockCargaRepo = {
      findCursoSeccionById: jest.fn().mockImplementation((id: string) => {
        if (id === "cs-mat-5a") return Promise.resolve(mockCursoSeccionMatematica);
        return Promise.resolve(null);
      }),
      findBySeccion: jest.fn().mockResolvedValue([mockItemMatematicaAsignado]),
      horasAsignadasDocente: jest.fn().mockResolvedValue(0),
      updateDocente: jest.fn().mockResolvedValue(mockItemMatematicaAsignado),
      findByDocente: jest.fn().mockResolvedValue([
        {
          cursoSeccionId: "cs-mat-5a",
          cursoNombre: "Matemática",
          seccionId: "sec-5a",
          gradoNumero: 5,
          seccionNombre: "A",
          horas: 6,
        },
      ]),
    };

    service = new CargaAcademicaService(
      mockCargaRepo as unknown as CargaAcademicaRepository,
      mockDocentesRepo as unknown as DocentesRepository,
      mockAniosService as unknown as AniosAcademicosService,
      mockAnioLectivoCls as unknown as AnioLectivoCls,
      mockAuditService as unknown as AuditService,
    );
  });

  describe("Asignación de docentes y control de jornada", () => {
    it("CP-RF008-06: debe asignar un docente al curso de la sección y reportar su nueva carga horaria", async () => {
      // Dado que 5.° A tiene Matemática sin docente
      mockCargaRepo.horasAsignadasDocente = jest.fn().mockResolvedValue(18);

      const dto: AsignarDocenteDto = {
        docenteId: "doc-quispe",
      };

      // Cuando asigno al docente Quispe y guardo
      const resultado = await service.asignarDocente("cs-mat-5a", dto, "user-admin", "192.168.1.50");

      // Entonces el sistema registra la asignación y muestra la nueva carga horaria del docente
      expect(resultado.item.docente?.id).toBe("doc-quispe");
      expect(resultado.cargaDocente.horasAsignadas).toBe(24); // 18 anteriores + 6 de Matemática
      expect(resultado.cargaDocente.jornadaHoras).toBe(30);
      expect(resultado.cargaDocente.excede).toBe(false);

      expect(mockCargaRepo.updateDocente).toHaveBeenCalledWith("cs-mat-5a", "doc-quispe");

      // Y audita la asignación
      expect(mockAuditService.log).toHaveBeenCalledWith(
        expect.objectContaining({
          usuarioId: "user-admin",
          accion: "actualizar",
          tabla: "cursos_seccion",
          registroId: "cs-mat-5a",
          direccionIp: "192.168.1.50",
        }),
      );
    });

    it("CP-RF008-07: debe bloquear la asignación de un docente en estado cesado o inactivo", async () => {
      // Dado que el docente Ramos está cesado
      const dto: AsignarDocenteDto = {
        docenteId: "doc-ramos",
      };

      // Cuando intento asignarlo a un curso
      // Entonces el servidor lo rechaza con "Solo se pueden asignar docentes activos."
      await expect(service.asignarDocente("cs-mat-5a", dto, "user-admin")).rejects.toThrow(
        new BadRequestException("Solo se pueden asignar docentes activos."),
      );
      expect(mockCargaRepo.updateDocente).not.toHaveBeenCalled();
    });

    it("CP-RF008-08: debe advertir exceso de jornada y bloquear si no hay confirmación; guardar con confirmarExceso", async () => {
      // Dado que el docente Quispe tiene jornada de 30 horas y 28 asignadas
      mockCargaRepo.horasAsignadasDocente = jest.fn().mockResolvedValue(28);

      const dtoSinConfirmar: AsignarDocenteDto = {
        docenteId: "doc-quispe",
        confirmarExceso: false,
      };

      // Cuando le asigno un curso de 6 horas sin confirmación
      // Entonces el sistema muestra "El docente supera su jornada: tendrá 34 horas asignadas de 30."
      await expect(
        service.asignarDocente("cs-mat-5a", dtoSinConfirmar, "user-admin"),
      ).rejects.toThrow(
        new ConflictException("El docente supera su jornada: tendrá 34 horas asignadas de 30."),
      );
      expect(mockCargaRepo.updateDocente).not.toHaveBeenCalled();

      // Cuando le asigno el curso con confirmarExceso: true
      const dtoConfirmado: AsignarDocenteDto = {
        docenteId: "doc-quispe",
        confirmarExceso: true,
      };
      const resultado = await service.asignarDocente("cs-mat-5a", dtoConfirmado, "user-admin");

      // Entonces el sistema guarda y reporta la carga con excede = true
      expect(resultado.cargaDocente.horasAsignadas).toBe(34);
      expect(resultado.cargaDocente.jornadaHoras).toBe(30);
      expect(resultado.cargaDocente.excede).toBe(true);
      expect(mockCargaRepo.updateDocente).toHaveBeenCalledWith("cs-mat-5a", "doc-quispe");
    });

    it("CP-RF008-09: debe cambiar el docente a mitad de año conservando las sesiones ya firmadas", async () => {
      // Dado que Matemática de 5.° A la dicta Quispe y ya tiene sesiones firmadas
      const dtoCambio: AsignarDocenteDto = {
        docenteId: "doc-flores",
      };

      // Cuando la asigno a la docente Flores
      const resultado = await service.asignarDocente("cs-mat-5a", dtoCambio, "user-admin");

      // Entonces el repositorio de carga se actualiza con Flores y no se alteran sesiones_clase
      expect(resultado.item).toBeDefined();
      expect(mockCargaRepo.updateDocente).toHaveBeenCalledWith("cs-mat-5a", "doc-flores");
      expect(mockAuditService.log).toHaveBeenCalledWith(
        expect.objectContaining({
          accion: "actualizar",
          tabla: "cursos_seccion",
          registroId: "cs-mat-5a",
        }),
      );
    });

    it("debe permitir desasignar un docente (docenteId: null)", async () => {
      const dtoQuitar: AsignarDocenteDto = {
        docenteId: null,
      };

      await service.asignarDocente("cs-mat-5a", dtoQuitar, "user-admin");

      expect(mockCargaRepo.updateDocente).toHaveBeenCalledWith("cs-mat-5a", null);
      expect(mockAuditService.log).toHaveBeenCalledWith(
        expect.objectContaining({
          accion: "actualizar",
          tabla: "cursos_seccion",
          registroId: "cs-mat-5a",
        }),
      );
    });
  });

  describe("Consultas de carga académica", () => {
    it("CP-RF008-10: debe listar los cursos y secciones con sus horas y el total frente a la jornada del docente", async () => {
      // Dado que estoy en la ficha del docente Quispe con 6 horas asignadas
      const cursosDocente: CursoDocenteItemDto[] = [
        {
          cursoSeccionId: "cs-mat-5a",
          cursoNombre: "Matemática",
          seccionId: "sec-5a",
          gradoNumero: 5,
          seccionNombre: "A",
          horas: 6,
        },
      ];
      mockCargaRepo.findByDocente = jest.fn().mockResolvedValue(cursosDocente);

      // Cuando abro "Carga académica"
      const resultado = await service.findByDocente("doc-quispe");

      // Entonces el sistema lista sus cursos y secciones con las horas y el total frente a su jornada
      expect(resultado.docenteId).toBe("doc-quispe");
      expect(resultado.jornadaHoras).toBe(30);
      expect(resultado.horasAsignadas).toBe(6);
      expect(resultado.excede).toBe(false);
      expect(resultado.cursos).toHaveLength(1);
      expect(resultado.cursos[0].cursoNombre).toBe("Matemática");
      expect(resultado.cursos[0].seccionNombre).toBe("A");
      expect(mockCargaRepo.findByDocente).toHaveBeenCalledWith("anio-2027", "doc-quispe");
    });

    it("debe consultar la carga académica de una sección completa", async () => {
      const resultado = await service.findBySeccion("sec-5a");

      expect(resultado.seccionId).toBe("sec-5a");
      expect(resultado.items).toHaveLength(1);
      expect(mockCargaRepo.findBySeccion).toHaveBeenCalledWith("sec-5a");
    });

    it("debe lanzar NotFoundException si el curso-sección no existe al asignar docente", async () => {
      mockCargaRepo.findCursoSeccionById = jest.fn().mockResolvedValue(null);

      await expect(
        service.asignarDocente("cs-inexistente", { docenteId: "doc-quispe" }, "user-admin"),
      ).rejects.toThrow(NotFoundException);
    });
  });
});
